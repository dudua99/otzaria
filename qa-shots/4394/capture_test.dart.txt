Link _l(String path2, String label, int at) => Link(
  heRef: '$path2 א, $label',
  index1: 1,
  path2: path2,
  index2: 1,
  connectionType: 'commentary',
  anchorStart: at,
  anchorLabel: label,
);

const _line =
    'יתגבר כארי לעמוד בבוקר לעבודת בוראו, שיהא הוא מעורר השחר. הגה: ועל כל פנים לא יאחר זמן התפלה שהצבור מתפללין.';

void main() {
  setUpAll(() async {
    await Settings.init(cacheProvider: MemorySettingsCache());
    final manifest =
        jsonDecode(await rootBundle.loadString('FontManifest.json')) as List;
    for (final entry in manifest.cast<Map<String, dynamic>>()) {
      final family = entry['family'] as String;
      if (family.contains('Icons')) continue;
      final loader = FontLoader(family);
      for (final f in (entry['fonts'] as List).cast<Map<String, dynamic>>()) {
        loader.addFont(rootBundle.load(f['asset'] as String));
      }
      await loader.load();
    }
  });
  final links = [
    _l('באר הגולה על שולחן ערוך אורח חיים', 'א', 6),
    _l('טורי זהב על שולחן ערוך אורח חיים', 'ב', 30),
    _l('מגן אברהם על שולחן ערוך אורח חיים', 'ג', 56),
    _l('באר הגולה על שולחן ערוך אורח חיים', 'ד', 107),
  ];
  for (final printLike in [false, true]) {
    for (final b in Brightness.values) {
      for (final w in [1000.0, 360.0]) {
        final name = '${printLike ? 'after' : 'before'}/${b.name}_w${w.toInt()}';
        testWidgets(name, (tester) async {
          final html = injectLinkAnchorMarkers(
            rawLine: _line,
            anchorLinks: links,
            styleIndexByCommentator: anchorStyleIndexByCommentator(links),
            lineIndex: 0,
            printLike: printLike,
          );
          final profile = TextDisplayProfile(
            anchorMarkerStyle:
                printLike ? AnchorMarkerStyle.print : AnchorMarkerStyle.variants,
          );
          await pumpGolden(
            tester,
            SizedBox(
              width: w - 16,
              child: Builder(
                builder: (context) => Column(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    Padding(
                      padding: const EdgeInsets.all(12),
                      child: SmartTextWidget(
                        text: html,
                        settings: const RenderSettings(
                          fontSize: 26,
                          fontFamily: 'FrankRuhlCLM',
                        ),
                        onAnchorTap: (_) {},
                      ),
                    ),
                    SettingsCard(
                      children: TextDisplayProfileEditor.tiles(
                        context,
                        profile: profile,
                        onChanged: (_) {},
                      ).sublist(5),
                    ),
                  ],
                ),
              ),
            ),
            brightness: b,
            size: Size(w, 620),
          );
          final boundary = tester.renderObject<RenderRepaintBoundary>(
            find.byKey(goldenBoundaryKey),
          );
          final bytes = await tester.runAsync(() async {
            final image = await boundary.toImage(pixelRatio: 2);
            final data = await image.toByteData(format: ui.ImageByteFormat.png);
            return data!.buffer.asUint8List();
          });
          final f = File('/tmp/cap4394/$name.png')..createSync(recursive: true);
          f.writeAsBytesSync(bytes!);
        });
      }
    }
  }
}
