import '../test_helpers/stub_settings_bloc.dart';
import 'golden_harness.dart';

void main() {
  const out = String.fromEnvironment('OUT');
  for (final b in [Brightness.light, Brightness.dark]) {
    testWidgets('cap $b', (tester) async {
      await Settings.init(cacheProvider: MemoryCacheProvider());
      await tester.binding.setSurfaceSize(const Size(900, 1100));
      final book = TextBook(title: 'ספר א');
      final bloc = TextBookBloc(
        repository: TextBookRepository(fileSystem: FileSystemData.instance),
        initialState: TextBookInitial.named(book, 0, false, const []),
      )..emit(
          TextBookLoaded(
            book: book,
            showLeftPane: false,
            content: const ['א'],
            fontSize: 25,
            /*AFTER*/commentatorsFontSize: 30,
            /*AFTER*/lineHeight: 2.2,
            showSplitView: false,
            activeCommentators: const [],
            commentatorGroups: const [],
            availableCommentators: const [],
            links: const [],
            linksByLine: const {},
            tableOfContents: const [],
            removeNikud: false,
            visibleIndices: const [0],
            pinLeftPane: false,
            searchText: '',
          ),
        );
      await tester.pumpWidget(
        RepaintBoundary(
          key: const ValueKey('cap'),
          child: MaterialApp(
            debugShowCheckedModeBanner: false,
            theme: goldenTheme(b),
            locale: const Locale('he'),
            supportedLocales: const [Locale('he')],
            localizationsDelegates: GlobalMaterialLocalizations.delegates,
            home: BlocProvider<SettingsBloc>.value(
              value: StubSettingsBloc(
                SettingsState.initial().copyWith(enablePerBookSettings: true),
              ),
              child: Scaffold(
                body: PrimaryScrollController.none(
                  child: TextSettingsTab(
                    isDialog: true,
                    scope: TextSettingsScope.book,
                    /*AFTER*/book: bloc,
                  ),
                ),
              ),
            ),
          ),
        ),
      );
      await tester.pump();
      final ro = tester.renderObject(find.byKey(const ValueKey('cap')))
          as RenderRepaintBoundary;
      await tester.runAsync(() async {
        final img = await ro.toImage(pixelRatio: 1);
        final bytes = await img.toByteData(format: ui.ImageByteFormat.png);
        File('$out/${b.name}_w900.png')
          ..createSync(recursive: true)
          ..writeAsBytesSync(bytes!.buffer.asUint8List());
      });
      await bloc.close();
    });
  }
}
