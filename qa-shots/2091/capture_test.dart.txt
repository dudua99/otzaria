import 'dart:io';
import 'dart:ui' as ui;

import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter_localizations/flutter_localizations.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:otzaria/external_catalog/repository/external_catalog_repository.dart';
import 'package:otzaria/library/view/grid_items.dart';
import 'package:otzaria/models/books.dart';
import 'package:otzaria/settings/engine/app_settings_store.dart';
import 'package:otzaria/settings/engine/settings_repository.dart';
import 'package:path/path.dart' as path;
import 'package:sqlite3/sqlite3.dart';

import '../test_helpers/memory_cache_provider.dart';
import '../support/golden_fonts.dart';
import 'golden_harness.dart';

void main() {
  const out = String.fromEnvironment('OUT');
  late List<ExternalLibraryBook> books;
  setUpAll(() async {
    final tmp = Directory.systemTemp.createTempSync('cap2091');
    Directory(path.join(tmp.path, 'Otzaria')).createSync(recursive: true);
    await Settings.init(cacheProvider: MemoryCacheProvider());
    await Settings.setValue<String>(SettingsRepository.keyLibraryPath, tmp.path);
    await Settings.setValue<String>(
      SettingsRepository.keyLibraryFolderName,
      'Otzaria',
    );
    final repo = ExternalCatalogRepository();
    final db = sqlite3.open(repo.databasePath);
    db.execute(
      'CREATE TABLE otzar_hahochma (book_id INTEGER PRIMARY KEY, title TEXT, '
      'authors TEXT, volume TEXT, from_year TEXT, to_year TEXT, places TEXT, '
      'subjects TEXT)',
    );
    const vols = [
      'אורח חיים א',
      'אורח חיים ב',
      'יורה דעה א',
      'יורה דעה ב',
      'אבן העזר',
      'חושן משפט',
    ];
    for (var i = 0; i < vols.length; i++) {
      db.execute(
        "INSERT INTO otzar_hahochma VALUES (?, 'ערוך השלחן', "
        "'[\"אפשטיין, יחיאל מיכל\"]', ?, '1903', NULL, 'ורשה', NULL)",
        [i + 1, vols[i]],
      );
    }
    db.close();
    books = (await repo.getOtzarBooks())..sort((a, b) => a.id!.compareTo(b.id!));
  });

  for (final b in [Brightness.light, Brightness.dark]) {
    for (final w in [360.0, 900.0]) {
      testWidgets('cap $b $w', (tester) async {
        await tester.runAsync(loadGoldenFonts);
        await tester.binding.setSurfaceSize(Size(w, 560));
        await tester.pumpWidget(
          RepaintBoundary(
            key: const ValueKey('cap'),
            child: MaterialApp(
              debugShowCheckedModeBanner: false,
              theme: goldenTheme(b),
              locale: const Locale('he'),
              supportedLocales: const [Locale('he')],
              localizationsDelegates: GlobalMaterialLocalizations.delegates,
              home: Scaffold(
                body: GridView.count(
                  crossAxisCount: w < 600 ? 1 : 3,
                  childAspectRatio: w < 600 ? 4 : 2.6,
                  padding: const EdgeInsets.all(8),
                  mainAxisSpacing: 8,
                  crossAxisSpacing: 8,
                  children: [
                    for (final book in books)
                      BookGridItem(book: book, onBookClickCallback: () {}),
                  ],
                ),
              ),
            ),
          ),
        );
        await tester.pump();
        final ro = tester.renderObject(find.byKey(const ValueKey('cap')))
            as RenderRepaintBoundary;
        await tester.runAsync(() async {
          final img = await ro.toImage(pixelRatio: 1);
          final bytes = await img.toByteData(format: ui.ImageByteFormat.png);
          File('$out/${b.name}_w${w.toInt()}.png')
            ..createSync(recursive: true)
            ..writeAsBytesSync(bytes!.buffer.asUint8List());
        });
      });
    }
  }
}
