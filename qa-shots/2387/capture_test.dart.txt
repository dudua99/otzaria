import '../support/golden_fonts.dart';
import 'golden_harness.dart';

class _Repo extends AttachedLibrariesRepository {
  _Repo(this.items) : super(copyByDefault: false);
  final List<AttachedLibrary> items;
  final _c = StreamController<Set<String>>.broadcast();
  @override
  List<AttachedLibrary> get libraries => items;
  @override
  List<String> get folders => const [];
  @override
  Stream<Set<String>> get changes => _c.stream;
}

AttachedLibrary _lib(String name, AttachedLibraryPlacement p) => AttachedLibrary(
  slug: name,
  displayName: name,
  path: 'C:/dbs/$name.db',
  mode: AttachedLibraryMode.link,
  status: AttachedLibraryStatus.ok,
  placement: p,
  bookCount: 128,
  capabilities: const {
    AttachedLibraryCapability.toc,
    AttachedLibraryCapability.categories,
  },
  addedAt: DateTime(2026),
);

void main() {
  final out = Platform.environment['QA_OUT']!;
  setUpAll(() async {
    await Settings.init(cacheProvider: MemorySettingsCache());
  });
  for (final b in Brightness.values) {
    for (final w in [1000.0, 360.0]) {
      for (final scale in [1.0, 1.3]) {
        final name = '${b.name}_w${w.toInt()}_x$scale';
        testWidgets(name, (tester) async {
          debugDefaultTargetPlatformOverride = goldenPlatform;
          tester.view.physicalSize = Size(w, 760);
          tester.view.devicePixelRatio = 1;
          tester.platformDispatcher.textScaleFactorTestValue = scale;
          addTearDown(tester.view.reset);
          addTearDown(tester.platformDispatcher.clearTextScaleFactorTestValue);
          final repo = _Repo([
            _lib('אוצר הספרים', AttachedLibraryPlacement.separateRoot),
            _lib('ספריית בית', AttachedLibraryPlacement.mergeIntoLibrary),
          ]);
          final bloc = AttachedLibrariesBloc(
            addLibraryEvent: (_) {},
            repository: repo,
          );
          addTearDown(bloc.close);
          const key = ValueKey('cap');
          await tester.pumpWidget(
            MaterialApp(
              debugShowCheckedModeBanner: false,
              theme: goldenTheme(b),
              home: Directionality(
                textDirection: TextDirection.rtl,
                child: Scaffold(
                  body: SingleChildScrollView(
                    child: RepaintBoundary(
                      key: key,
                      child: Material(
                        child: BlocProvider.value(
                          value: bloc,
                          child: const AttachedLibrariesPanel(
                            supportsLinking: true,
                          ),
                        ),
                      ),
                    ),
                  ),
                ),
              ),
            ),
          );
          await tester.pump();
          await tester.pump(const Duration(milliseconds: 300));
          final ex = tester.takeException();
          final boundary =
              tester.renderObject(find.byKey(key)) as RenderRepaintBoundary;
          await tester.runAsync(() async {
            final img = await boundary.toImage(pixelRatio: 2);
            final bytes = await img.toByteData(format: ui.ImageByteFormat.png);
            File('$out/$name.png')
              ..createSync(recursive: true)
              ..writeAsBytesSync(bytes!.buffer.asUint8List());
          });
          debugDefaultTargetPlatformOverride = null;
          if (ex != null) fail('exception: $ex');
        });
      }
    }
  }
}
